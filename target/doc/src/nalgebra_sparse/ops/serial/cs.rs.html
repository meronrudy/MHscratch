15/// Helper functionality for implementing CSR/CSC SPMM.
16///
17/// Since CSR/CSC matrices are basically transpositions of each other, which lets us use the same
18/// algorithm for the SPMM implementation. The implementation here is written in a CSR-centric
19/// manner. This means that when using it for CSC, the order of the matrices needs to be
20/// reversed (since transpose(AB) = transpose(B) * transpose(A) and CSC(A) = transpose(CSR(A)).
21///
22/// We assume here that the matrices have already been verified to be dimensionally compatible.
23pub fn spmm_cs_prealloc_unchecked<T>(
24    beta: T,
25    c: &mut CsMatrix<T>,
26    alpha: T,
27    a: &CsMatrix<T>,
28    b: &CsMatrix<T>,
29) -> Result<(), OperationError>
30where
31    T: Scalar + ClosedAddAssign + ClosedMulAssign + Zero + One,
32{
33    assert_eq!(c.pattern().major_dim(), a.pattern().major_dim());
34    assert_eq!(c.pattern().minor_dim(), b.pattern().minor_dim());
35    let some_val = Zero::zero();
36    let mut scratchpad_values: Vec<T> = vec![some_val; b.pattern().minor_dim()];
37    for i in 0..c.pattern().major_dim() {
38        let a_lane_i = a.get_lane(i).unwrap();
39
40        let mut c_lane_i = c.get_lane_mut(i).unwrap();
41
42        for (&k, a_ik) in a_lane_i.minor_indices().iter().zip(a_lane_i.values()) {
43            let b_lane_k = b.get_lane(k).unwrap();
44            let alpha_aik = alpha.clone() * a_ik.clone();
45            for (j, b_kj) in b_lane_k.minor_indices().iter().zip(b_lane_k.values()) {
46                // use a dense scatter vector to accumulate non-zeros quickly
47                unsafe {
48                    *scratchpad_values.get_unchecked_mut(*j) += alpha_aik.clone() * b_kj.clone();
49                }
50            }
51        }
52
53        //Get indices from C pattern and gather from the dense scratchpad_values
54        let (indices, values) = c_lane_i.indices_and_values_mut();
55        values
56            .iter_mut()
57            .zip(indices)
58            .for_each(|(output_ref, index)| unsafe {
59                *output_ref = beta.clone() * output_ref.clone()
60                    + scratchpad_values.get_unchecked(*index).clone();
61                *scratchpad_values.get_unchecked_mut(*index) = Zero::zero();
62            });
63    }
64
65    Ok(())
66}

68pub fn spmm_cs_prealloc<T>(
69    beta: T,
70    c: &mut CsMatrix<T>,
71    alpha: T,
72    a: &CsMatrix<T>,
73    b: &CsMatrix<T>,
74) -> Result<(), OperationError>
75where
76    T: Scalar + ClosedAddAssign + ClosedMulAssign + Zero + One,
77{
78    for i in 0..c.pattern().major_dim() {
79        let a_lane_i = a.get_lane(i).unwrap();
80        let mut c_lane_i = c.get_lane_mut(i).unwrap();
81        for c_ij in c_lane_i.values_mut() {
82            *c_ij = beta.clone() * c_ij.clone();
83        }
84
85        for (&k, a_ik) in a_lane_i.minor_indices().iter().zip(a_lane_i.values()) {
86            let b_lane_k = b.get_lane(k).unwrap();
87            let (mut c_lane_i_cols, mut c_lane_i_values) = c_lane_i.indices_and_values_mut();
88            let alpha_aik = alpha.clone() * a_ik.clone();
89            for (j, b_kj) in b_lane_k.minor_indices().iter().zip(b_lane_k.values()) {
90                // Determine the location in C to append the value
91                let (c_local_idx, _) = c_lane_i_cols
92                    .iter()
93                    .enumerate()
94                    .find(|(_, c_col)| *c_col == j)
95                    .ok_or_else(spmm_cs_unexpected_entry)?;
96
97                c_lane_i_values[c_local_idx] += alpha_aik.clone() * b_kj.clone();
98                c_lane_i_cols = &c_lane_i_cols[c_local_idx..];
99                c_lane_i_values = &mut c_lane_i_values[c_local_idx..];
100            }
101        }
102    }
103
104    Ok(())
105}

114/// Helper functionality for implementing CSR/CSC SPADD.
115pub fn spadd_cs_prealloc<T>(
116    beta: T,
117    c: &mut CsMatrix<T>,
118    alpha: T,
119    a: Op<&CsMatrix<T>>,
120) -> Result<(), OperationError>
121where
122    T: Scalar + ClosedAddAssign + ClosedMulAssign + Zero + One,
123{
124    match a {
125        Op::NoOp(a) => {
126            for (mut c_lane_i, a_lane_i) in c.lane_iter_mut().zip(a.lane_iter()) {
127                if beta != T::one() {
128                    for c_ij in c_lane_i.values_mut() {
129                        *c_ij *= beta.clone();
130                    }
131                }
132
133                let (mut c_minors, mut c_vals) = c_lane_i.indices_and_values_mut();
134                let (a_minors, a_vals) = (a_lane_i.minor_indices(), a_lane_i.values());
135
136                for (a_col, a_val) in a_minors.iter().zip(a_vals) {
137                    // TODO: Use exponential search instead of linear search.
138                    // If C has substantially more entries in the row than A, then a line search
139                    // will needlessly visit many entries in C.
140                    let (c_idx, _) = c_minors
141                        .iter()
142                        .enumerate()
143                        .find(|(_, c_col)| *c_col == a_col)
144                        .ok_or_else(spadd_cs_unexpected_entry)?;
145                    c_vals[c_idx] += alpha.clone() * a_val.clone();
146                    c_minors = &c_minors[c_idx..];
147                    c_vals = &mut c_vals[c_idx..];
148                }
149            }
150        }
151        Op::Transpose(a) => {
152            if beta != T::one() {
153                for c_ij in c.values_mut() {
154                    *c_ij *= beta.clone();
155                }
156            }
157
158            for (i, a_lane_i) in a.lane_iter().enumerate() {
159                for (&j, a_val) in a_lane_i.minor_indices().iter().zip(a_lane_i.values()) {
160                    let a_val = a_val.clone();
161                    let alpha = alpha.clone();
162                    match c.get_entry_mut(j, i).unwrap() {
163                        SparseEntryMut::NonZero(c_ji) => *c_ji += alpha * a_val,
164                        SparseEntryMut::Zero => return Err(spadd_cs_unexpected_entry()),
165                    }
166                }
167            }
168        }
169    }
170    Ok(())
171}

173/// Helper functionality for implementing CSR/CSC SPMM.
174///
175/// The implementation essentially assumes that `a` is a CSR matrix. To use it with CSC matrices,
176/// the transposed operation must be specified for the CSC matrix.
177pub fn spmm_cs_dense<T>(
178    beta: T,
179    mut c: DMatrixViewMut<'_, T>,
180    alpha: T,
181    a: Op<&CsMatrix<T>>,
182    b: Op<DMatrixView<'_, T>>,
183) where
184    T: Scalar + ClosedAddAssign + ClosedMulAssign + Zero + One,
185{
186    match a {
187        Op::NoOp(a) => {
188            for j in 0..c.ncols() {
189                let mut c_col_j = c.column_mut(j);
190                for (c_ij, a_row_i) in c_col_j.iter_mut().zip(a.lane_iter()) {
191                    let mut dot_ij = T::zero();
192                    for (&k, a_ik) in a_row_i.minor_indices().iter().zip(a_row_i.values()) {
193                        let b_contrib = match b {
194                            Op::NoOp(ref b) => b.index((k, j)),
195                            Op::Transpose(ref b) => b.index((j, k)),
196                        };
197                        dot_ij += a_ik.clone() * b_contrib.clone();
198                    }
199                    *c_ij = beta.clone() * c_ij.clone() + alpha.clone() * dot_ij;
200                }
201            }
202        }
203        Op::Transpose(a) => {
204            // In this case, we have to pre-multiply C by beta
205            c *= beta;
206
207            for k in 0..a.pattern().major_dim() {
208                let a_row_k = a.get_lane(k).unwrap();
209                for (&i, a_ki) in a_row_k.minor_indices().iter().zip(a_row_k.values()) {
210                    let gamma_ki = alpha.clone() * a_ki.clone();
211                    let mut c_row_i = c.row_mut(i);
212                    match b {
213                        Op::NoOp(ref b) => {
214                            let b_row_k = b.row(k);
215                            for (c_ij, b_kj) in c_row_i.iter_mut().zip(b_row_k.iter()) {
216                                *c_ij += gamma_ki.clone() * b_kj.clone();
217                            }
218                        }
219                        Op::Transpose(ref b) => {
220                            let b_col_k = b.column(k);
221                            for (c_ij, b_jk) in c_row_i.iter_mut().zip(b_col_k.iter()) {
222                                *c_ij += gamma_ki.clone() * b_jk.clone();
223                            }
224                        }
225                    }
226                }
227            }
228        }
229    }
